{{- /* HowTo JSON-LD — single jsonify+safeJS to avoid auto-escape double-encoding */ -}} {{- $stepObjs := slice -}} {{- range $i, $s := $steps -}} {{- $stepObjs = $stepObjs | append (dict "@type" "HowToStep" "position" (add $i 1) "name" $s.name "text" ($s.text | markdownify | plainify)) -}} {{- end -}} {{- $h := dict "@context" "https://schema.org" "@type" "HowTo" "name" $title "step" $stepObjs -}} {{- with $duration }}{{ $h = merge $h (dict "totalTime" .) }}{{ end -}} {{- with $image }}{{ $h = merge $h (dict "image" (. | absURL)) }}{{ end -}} {{- end -}}
